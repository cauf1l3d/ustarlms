name: USTAR context secrets guard

# Не даёт секретам и внутренним адресам попасть в context/.
# Нужен потому что агент, выполняющий «обнови контекст», может закоммитить креды,
# если нигде не сказано «нельзя». Это защита от конкретного сценария, не общая практика.

on:
  pull_request:
    branches: [main]
    paths: ['context/**']
  push:
    branches: [main]
    paths: ['context/**']
  workflow_dispatch:

permissions:
  contents: read

jobs:
  no-secrets:
    runs-on: ubuntu-24.04
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}

      - name: Secret patterns in context
        run: |
          set +e
          hits=$(grep -rniE '(password|passwd|secret|api[_-]?key|token|BEGIN [A-Z ]*PRIVATE KEY)' \
                   context/ --include='*.txt' --include='*.json' --include='*.yaml' --include='*.yml' \
                 | grep -viE 'SET$|=-$|redacted|<internal>|not_configured' || true)
          if [ -n "$hits" ]; then
            echo "$hits"
            echo "::error::Возможный секрет в context/. Значения не коммитить: config.php — только SET/-, docker Env исключать."
            exit 1
          fi
          echo "OK: секретных паттернов не найдено"

      - name: Internal addresses in server-facts
        run: |
          set +e
          if [ -d context/runtime/server-facts ]; then
            hits=$(grep -rnE '\b(10|172\.(1[6-9]|2[0-9]|3[01])|192\.168)\.[0-9]+\.[0-9]+\b' \
                     context/runtime/server-facts/ || true)
            if [ -n "$hits" ]; then
              echo "$hits"
              echo "::error::Внутренний адрес в снимке. Заменить на <internal>."
              exit 1
            fi
          fi
          echo "OK: внутренних адресов не найдено"

      - name: config.php values must not be present
        run: |
          set +e
          # Разрешены только строки вида "ключ=SET" или "ключ=-".
          if grep -rn '\$CFG->' context/ 2>/dev/null | grep -vE '=SET|=(-)$' | grep -q .; then
            grep -rn '\$CFG->' context/ | grep -vE '=SET|=(-)$' | head
            echo "::error::Кажется, в context/ попали значения config.php. Допустимы только ключи SET/-."
            exit 1
          fi
          echo "OK: значений config.php нет"

      - name: Snapshot must carry manifest
        run: |
          set +e
          fail=0
          for d in context/runtime/server-facts/*/; do
            [ -d "$d" ] || continue
            [ -f "${d}_manifest.txt" ] || { echo "::error::Нет _manifest.txt в $d"; fail=1; }
            [ -f "${d}README.md" ]    || { echo "::error::Нет README.md в $d"; fail=1; }
          done
          exit $fail
